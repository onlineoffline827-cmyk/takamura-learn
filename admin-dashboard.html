import { createClient } from "npm:@supabase/supabase-js@2"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS" }; function jsonResponse(data: unknown, status = 200) { return new Response(JSON.stringify(data), { status, headers: { ...corsHeaders, "Content-Type": "application/json" } }); } Deno.serve(async (req) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } if (req.method !== "POST") { return jsonResponse({ error: "Only POST requests are allowed." }, 405); } try { // ENVIRONMENT VARIABLES const supabaseUrl = Deno.env.get("SUPABASE_URL"); const anonKey = Deno.env.get("SUPABASE_ANON_KEY"); const serviceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"); const ocrApiKey = Deno.env.get("OCR_API_KEY"); if (!supabaseUrl || !anonKey || !serviceKey || !ocrApiKey) { return jsonResponse({ error: "Required environment variables are missing." }, 500); } // VERIFY ADMIN SESSION const authHeader = req.headers.get("Authorization"); if (!authHeader || !authHeader.startsWith("Bearer ")) { return jsonResponse({ error: "Authentication required." }, 401); } const token = authHeader.replace("Bearer ", ""); const userClient = createClient( supabaseUrl, anonKey, { global: { headers: { Authorization: authHeader } }, auth: { persistSession: false, autoRefreshToken: false } } ); const { data: { user }, error: authError } = await userClient.auth.getUser(token); if (authError || !user) { return jsonResponse({ error: "Invalid user session." }, 401); } // CHECK ADMIN ROLE const { data: profile, error: profileError } = await userClient .from("profiles") .select("role") .eq("id", user.id) .single(); if (profileError || profile?.role !== "admin") { console.error("ADMIN CHECK ERROR:", profileError); return jsonResponse({ error: "Admin permission required." }, 403); } // READ UPLOAD ID const body = await req.json(); const uploadId = body.upload_id; if ( typeof uploadId !== "string" || !/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(uploadId) ) { return jsonResponse({ error: "A valid upload_id is required." }, 400); } // ADMIN CLIENT const adminClient = createClient( supabaseUrl, serviceKey, { auth: { persistSession: false, autoRefreshToken: false } } ); // GET DOCUMENT RECORD const { data: document, error: documentError } = await adminClient .from("lesson_uploads") .select( "id, file_path, file_name, mime_type, file_size, created_by, status" ) .eq("id", uploadId) .single(); // IMPROVED DOCUMENT ERROR DETAILS if (documentError || !document) { console.error("DOCUMENT LOOKUP FAILED:", { uploadId, error: documentError?.message, code: documentError?.code, details: documentError?.details, hint: documentError?.hint }); return jsonResponse({ error: "Uploaded document not found.", database_error: documentError?.message || null, database_code: documentError?.code || null, database_details: documentError?.details || null, requested_id: uploadId }, 404); } // VERIFY DOCUMENT OWNER if (document.created_by !== user.id) { return jsonResponse({ error: "You cannot process this document." }, 403); } // CHECK FILE TYPE if ( ![ "application/pdf", "image/jpeg", "image/png", "image/webp" ].includes(document.mime_type) ) { return jsonResponse({ error: "Unsupported document type." }, 400); } // CHECK OCR FILE SIZE if (Number(document.file_size) > 1024 * 1024) { return jsonResponse({ error: "OCR.space free plan supports files up to 1 MB." }, 413); } // DOWNLOAD ORIGINAL FILE const { data: file, error: downloadError } = await adminClient.storage .from("lesson-files") .download(document.file_path); if (downloadError || !file) { throw new Error( "Could not download uploaded document: " + (downloadError?.message || "File not found.") ); } // UPDATE STATUS const { error: statusError } = await adminClient .from("lesson_uploads") .update({ status: "processing" }) .eq("id", uploadId); if (statusError) throw statusError; // PREPARE OCR REQUEST const formData = new FormData(); const fileBlob = new Blob( [await file.arrayBuffer()], { type: document.mime_type } ); formData.append( "file", fileBlob, document.file_name ); formData.append("language", "auto"); formData.append("isOverlayRequired", "false"); formData.append("detectOrientation", "true"); formData.append("scale", "true"); formData.append("OCREngine", "2"); if (document.mime_type === "application/pdf") { formData.append("filetype", "PDF"); } // SEND TO OCR.SPACE const ocrResponse = await fetch( "https://api.ocr.space/parse/image", { method: "POST", headers: { apikey: ocrApiKey }, body: formData } ); const ocrResult = await ocrResponse.json(); if (!ocrResponse.ok || ocrResult.IsErroredOnProcessing) { const errorMessage = Array.isArray(ocrResult.ErrorMessage) ? ocrResult.ErrorMessage.join(", ") : ocrResult.ErrorMessage || "OCR processing failed."; await adminClient .from("lesson_uploads") .update({ status: "ocr_failed" }) .eq("id", uploadId); return jsonResponse({ error: errorMessage }, 502); } // EXTRACT ORIGINAL TEXT const pages = ocrResult.ParsedResults || []; const extractedText = pages .map((page: { ParsedText?: string }) => page.ParsedText || "" ) .join("\n\n"); if (!extractedText.trim()) { await adminClient .from("lesson_uploads") .update({ status: "ocr_failed" }) .eq("id", uploadId); return jsonResponse({ error: "No readable text was found in this document." }, 422); } // SAVE EXTRACTED TEXT const { error: saveError } = await adminClient .from("lesson_uploads") .update({ extracted_text: extractedText, status: "ocr_completed" }) .eq("id", uploadId); if (saveError) throw saveError; return jsonResponse({ success: true, message: "OCR completed successfully.", upload_id: uploadId, pages_processed: pages.length, text_length: extractedText.length, status: "ocr_completed" }); } catch (error) { console.error("OCR FUNCTION ERROR:", error); return jsonResponse({ error: error instanceof Error ? error.message : "Unexpected OCR error." }, 500); } });
